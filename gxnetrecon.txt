#!/usr/bin/env python3

import argparse
import json
import socket


def service_name(port):
    try:
        return socket.getservbyport(
            port,
            "tcp"
        )
    except OSError:
        return "unknown"


def parse_ports(value):
    ports = []

    for item in value.split(","):
        item = item.strip()

        if not item:
            continue

        try:
            port = int(item)
        except ValueError:
            raise ValueError(
                f"Puerto inválido: {item}"
            )

        if not 1 <= port <= 65535:
            raise ValueError(
                f"Puerto fuera de rango: {port}"
            )

        ports.append(port)

    if not ports:
        raise ValueError(
            "No se especificaron puertos"
        )

    return sorted(set(ports))


def main():
    parser = argparse.ArgumentParser(
        description="GxNetRecon - reconocimiento TCP autorizado"
    )

    parser.add_argument("target")
    parser.add_argument(
        "--ports",
        default="22,53,80,443,8080"
    )
    parser.add_argument(
        "--timeout",
        type=float,
        default=1.0
    )
    parser.add_argument(
        "-o",
        "--output",
        default="gxnetrecon_report.json"
    )

    args = parser.parse_args()

    if args.timeout <= 0:
        raise SystemExit(
            "[!] El timeout debe ser mayor que 0"
        )

    try:
        ip = socket.gethostbyname(
            args.target
        )

        ports = parse_ports(
            args.ports
        )

    except (socket.gaierror, ValueError) as error:
        raise SystemExit(
            f"[!] {error}"
        )

    open_ports = []

    for port in ports:

        with socket.socket(
            socket.AF_INET,
            socket.SOCK_STREAM
        ) as sock:

            sock.settimeout(
                args.timeout
            )

            if sock.connect_ex(
                (ip, port)
            ) == 0:

                service = service_name(
                    port
                )

                open_ports.append({
                    "port": port,
                    "service": service
                })

                print(
                    f"[OPEN] {port} ({service})"
                )

    report = {
        "tool": "GxNetRecon",
        "target": args.target,
        "ip": ip,
        "open_ports": open_ports,
    }

    with open(
        args.output,
        "w",
        encoding="utf-8"
    ) as f:
        json.dump(
            report,
            f,
            indent=2,
            ensure_ascii=False
        )

    print(
        f"[+] Reporte: {args.output}"
    )

    main()
